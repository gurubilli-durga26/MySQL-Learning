----------Operators in MySQL---------

1.Arithmetic Operators

Operator	Meaning	Example
+	Addition	salary + 5000
-	Subtraction	salary - 5000
*	Multiplication	salary * 12
/	Division	salary / 12
%	Modulus  	salary % 2

SELECT salary + 5000 AS new_salary
FROM employees;

SELECT salary * 12 AS annual_salary
FROM employees;

SELECT salary % 2 AS remainder
FROM employees;

2.Comparison Operators

SELECT *
FROM employees
WHERE salary > 50000;

SELECT *
FROM employees
WHERE salary = 50000;

SELECT *
FROM employees
WHERE salary != 50000;

3.Logical Operators

--> AND

SELECT *
FROM employees
WHERE department = 'IT'
AND salary > 50000;

--> OR

SELECT *
FROM employees
WHERE department = 'IT'
OR department = 'HR';

--> NOT

SELECT *
FROM employees
WHERE NOT department = 'IT';

4. BETWEEN

SELECT *
FROM table_name
WHERE column_name BETWEEN value1 AND value2;

SELECT *
FROM employees
WHERE salary BETWEEN 40000 AND 60000;

5. IN Operator

SELECT *
FROM employees
WHERE department IN ('IT', 'HR', 'Finance');

6. NOT IN Operator

SELECT *
FROM employees
WHERE department NOT IN ('IT', 'HR');

7.LIKE Operator

LIKE is used for pattern matching, especially with strings.

There are two important wildcards:

% → Any number of characters
_ → Exactly one character

'R%'    → starts with R
'%R'    → ends with R
'%R%'   → contains R anywhere

SELECT *
FROM employees
WHERE emp_name LIKE 'R%';

SELECT *
FROM employees
WHERE emp_name LIKE '%a';

SELECT *
FROM employees
WHERE emp_name LIKE '%vi%';

SELECT *
FROM employees
WHERE emp_name LIKE 'Ra___';

8. IS NULL Operator

NULL means no value / missing value.

SELECT *
FROM employees
WHERE city IS NULL;

SELECT *
FROM employees
WHERE city IS NOT NULL;

9. Assignment Operator

UPDATE employees
SET salary = 60000
WHERE emp_id = 101;

10. Operator Precedence
MUL OPERATORS ARE USED

SELECT *
FROM employees
WHERE department = 'IT'
AND salary > 50000
OR city = 'Hyderabad';
